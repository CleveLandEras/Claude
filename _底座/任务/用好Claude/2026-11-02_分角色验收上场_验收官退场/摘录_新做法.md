```python
# 用法：python3 读成品.py <pptx 路径>   打印每页的文字框、表格（逐行）和备注
import sys
from pptx import Presentation
p = Presentation(sys.argv[1])
print("页数", len(p.slides))
for i, s in enumerate(p.slides, 1):
    print(f"===== 第 {i} 页")
    for sh in s.shapes:
        if sh.has_table:
            print("[表格]")
            for r in sh.table.rows:
                print("  " + " | ".join(c.text.replace("\n", " ") for c in r.cells))
        elif sh.has_text_frame and sh.text_frame.text.strip():
            print("[文字] " + sh.text_frame.text.replace("\n", " / "))
    if s.has_notes_slide and s.notes_slide.notes_text_frame.text.strip():
        print("[备注]\n" + s.notes_slide.notes_text_frame.text)
    else:
        print("[备注] 无")
```

### 10.5 摘录是怎么从任务卡生成的（`生成摘录.py` 全文，分发对话草稿目录；它决定评审能看到卡的哪几条、去掉哪几条）

```python
# 用法：python3 生成摘录.py  —— 从三份任务文件的对齐卡重新生成给评审的摘录（验收材料/对齐卡_*.md），保留原摘录里「打分四档」之后的部分
import re, os
T = os.path.expanduser("~/claude/_底座/任务/弹窗标签推送系统/")
M = os.path.dirname(os.path.abspath(__file__)) + "/验收材料/"
KEEP = ("目标", "内容", "要做的", "读者", "可用时间", "我们是谁", "已替 Dk 定", "完成标准", "按人群筛选")
def card_bullets(fn):
    s = open(T + fn, encoding="utf-8").read()
    s = s[s.find("## 对齐卡"):]; s = s[s.find("\n")+1:]; s = s[:s.find("\n## ")]
    out, cur = [], None
    for line in s.split("\n"):
        m = re.match(r"^- \*\*(.+?)\*\*[:：](.*)$", line)
        if m:
            cur = [m.group(1), m.group(2), []]; out.append(cur)
        elif cur is not None and line.strip():
```
